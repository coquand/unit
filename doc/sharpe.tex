Let $R$ be a commutative ring and $F$ an $R$-module.  We consider a
linear system with coefficients in $R$ and unknowns in $F$,
\[
  x_1U_1+\dots+x_nU_n=V,\tag{1}
\]
where $U_1,\dots,U_n$ are column vectors of $R^m$ and $V\in F^m$.  Let
$A=(U_1\cdots U_n)$ and $B=(U_1\cdots U_n\,V)$, and write $\Delta_l(M)$
for the ideal (or, for $B$, the submodule of $F$) generated by the
$l\times l$ minors of $M$.  If $\Delta_n(A)$ is regular on $F$, (1) has at
most one solution, and a solution forces $\Delta_{n+1}(B)=0$.
Sharpe~\cite{Sharpe} proved that grade~$2$ gives existence.

\begin{theorem}[Sharpe]\label{thm:sharpe}
If $\Gr(\Delta_n(A);F)\geq2$ and $\Delta_{n+1}(B)=0$, the system (1) has
exactly one solution.  Conversely, if (1) has a solution, then
$\Delta_{n+1}(B)=0$.
\end{theorem}

Grade~$1$ does not suffice: over $\mathbb Z$ the system $2x=3$, $4x=6$
satisfies $\Delta_2(B)=0$, but the grade of $\langle2,4\rangle$ is only
$1$.  The theorem is Theorem B-7.5 of~\cite{CL}, where it is derived from
a local-global principle for coverings in depth~$2$.

The proof below is the one of the note~\cite{Co10}: the minors are
coordinates in the exterior algebra, Cramer's rule is applied to each
$n\times n$ subsystem, and the depth-$2$ hypothesis is used once, on the
Cramer numerators, which are proportional by uniqueness.  Here the
unknowns are in a module, as in Sharpe's paper, and the identity on which
everything rests (Lemma~\ref{lem:key}) is proved for all coordinates at
once by a contraction.  The theorem is checked in Unit
(\code{sag/depth2}).

\section{Setting}

Let $e_1,\dots,e_m$ be the canonical basis of $R^m$.  For a finite set
$I=\{i_1<\dots<i_k\}$ write $e_I=e_{i_1}\wedge\dots\wedge e_{i_k}$, and for
$x$ in the exterior algebra $\Lambda=\Lambda(R^m)$ write
$\co xI$ for its coordinate on $e_I$.  Let
\[
  \omega=U_1\wedge\dots\wedge U_n,\qquad
  P_k(b)=U_1\wedge\dots\wedge U_{k-1}\wedge b\wedge U_{k+1}\wedge\dots\wedge U_n
  \quad(b\in R^m).
\]
The coordinates $\delta_I=\co\omega I$ are the $n\times n$ minors of $A$, so
$\Delta_n(A)$ is generated by the family $(\delta_I)_I$.  For $F=R$ the
coordinates of $\omega\wedge V$ are the $(n+1)$-minors of $B$; in general we
put $\omega\wedge V=\sum_t(\omega\wedge e_t)\otimes V_t$, and its
coordinates
\[
  \textstyle\mu_I=\sum_t\co{\omega\wedge e_t}I\,V_t\in F
\]
generate $\Delta_{n+1}(B)$.  Cramer's numerators for the subsystem of the
rows $I$ are
\[
  \textstyle\lambda_k(I)=\sum_t\co{P_k(e_t)}I\,V_t\in F,
\]
which for $F=R$ is $\co{P_k(V)}I$: in coordinates,
$\delta_Ie_I=U_1(I)\wedge\dots\wedge U_n(I)$ and
$\lambda_k(I)e_I=U_1(I)\wedge\dots\wedge V(I)\wedge\dots\wedge U_n(I)$,
with $V(I)$ in place $k$ and $X(I)=\sum_{i\in I}X_ie_i$~\cite{Co10}.

Grade is used in an explicit form~\cite{CL,CQ}, equivalent for finitely
generated ideals to the true grade of Hochster and
Northcott~\cite{Northcott}.  For a finite family $(d_I)$ of elements of $R$:
$\Gr(d;F)\geq1$ if $d_Ix=0$ for all $I$ implies $x=0$ ($x\in F$); and
$\Gr(d;F)\geq2$ if moreover every family $(y_I)$ in $F$ with
$d_Iy_J=d_Jy_I$ for all $I,J$ is of the form $y_I=d_Ie$ with $e\in F$
(unique, by $\Gr\geq1$).

\section{Two lemmas}

\begin{lemma}[Cramer]\label{lem:cramer}
For $x\in F^n$ and every $I$ and $k$,
$\sum_t\co{P_k(e_t)}I\,(\textstyle\sum_jx_jU_j)_t=\delta_I\,x_k$.  Hence, if
$\Gr(\Delta_n(A);F)\geq1$, the system (1) has at most one solution.
\end{lemma}

\begin{proof}
For $F=R$ the left-hand side is $\co{P_k(\sum_jx_jU_j)}I$, since $P_k$ is
linear, and $P_k(\sum_jx_jU_j)=\sum_jx_jP_k(U_j)=x_k\,\omega$, because
$P_k(U_j)$ has the factor $U_j$ twice for $j\neq k$.  The general case is
the same computation with coefficients in $F$.  If $x$ and $y$ are
solutions, then $\delta_Ix_k=\delta_Iy_k$ for all $I$, so $x=y$.
\end{proof}

\begin{lemma}[key identity]\label{lem:key}
Let $\varphi$ be a linear form on $R^m$ and $\iota_\varphi$ the contraction
by $\varphi$ on $\Lambda$.  For every $b\in R^m$,
\[
  \textstyle\sum_j\varphi(U_j)\,P_j(b)=\varphi(b)\,\omega
  +(-1)^{n-1}\iota_\varphi(\omega\wedge b).
\]
In particular, if $\omega\wedge b=0$, then
$\sum_jU_j^r\,\co{P_j(b)}I=b_r\,\delta_I$ for every row $r$ and every $I$.
\end{lemma}

\begin{proof}
$\iota_\varphi$ is a derivation of degree $-1$:
$\iota_\varphi(w_1\wedge\dots\wedge w_k)=\sum_l(-1)^{l-1}\varphi(w_l)\,
w_1\wedge\dots\widehat{w_l}\dots\wedge w_k$.  Applied to
$U_1\wedge\dots\wedge U_n\wedge b$, the term $l=n+1$ is
$(-1)^n\varphi(b)\,\omega$, and for $l=j\leq n$ moving $b$ from the end
to place $j$ gives $U_1\wedge\dots\widehat{U_j}\dots\wedge U_n\wedge
b=(-1)^{n-j}P_j(b)$, so the term is $(-1)^{n-1}\varphi(U_j)P_j(b)$.  For the
second claim take $\varphi(w)=w_r$ and the coordinate on $e_I$.
\end{proof}

For $F=R$ the second claim is the identity proved in the appendix
of~\cite{Co10} for $i\notin I$, by expanding
$(V(i)+V(I))\wedge(U_1(i)+U_1(I))\wedge\dots\wedge(U_n(i)+U_n(I))=0$; the
contraction gives it for all $I$ and $r$ at once.

\section{Proof of the theorem}

Applying Lemma~\ref{lem:key} to $b=e_t$, multiplying by $V_t$ and summing
over $t$ gives, for every $I$ and $r$,
\[
  \textstyle\sum_jU_j^r\,\lambda_j(I)=\delta_I\,V_r+\sum_{I'}k_{I'}\,\mu_{I'},
  \tag{2}
\]
where $k_{I'}=(-1)^{n-1}\co{\iota_\varphi(e_{I'})}I$, $\varphi$ the $r$-th
coordinate: the last term of Lemma~\ref{lem:key} is linear in
$\omega\wedge e_t$, hence a combination of its coordinates.

\emph{Existence.}  Assume $\Delta_{n+1}(B)=0$, i.e.\ all $\mu_I=0$.  By
(2), $\lambda(I)=(\lambda_1(I),\dots,\lambda_n(I))$ satisfies
\[
  \textstyle\sum_j\lambda_j(I)\,U_j=\delta_I\,V\qquad\text{for every }I.
\]
For two subsets $I,J$ the vectors $\delta_J\lambda(I)$ and
$\delta_I\lambda(J)$ both solve the system with right-hand side
$\delta_I\delta_JV$, so they are equal by Lemma~\ref{lem:cramer}:
$\delta_J\lambda_k(I)=\delta_I\lambda_k(J)$ for every $k$.  Since
$\Gr(\Delta_n(A);F)\geq2$, there is $x_k\in F$ with
$\lambda_k(I)=\delta_Ix_k$ for all $I$.  Then
\[
  \textstyle\delta_I\sum_jx_jU_j=\sum_j\lambda_j(I)\,U_j=\delta_I\,V
\]
for all $I$, and $\sum_jx_jU_j=V$ since $\Delta_n(A)$ is regular on $F$.
Uniqueness is Lemma~\ref{lem:cramer}.

\emph{Converse.}  If $\sum_jx_jU_j=V$, then
$\mu_I=\sum_jx_j\,\co{\omega\wedge U_j}I=0$, since $\omega\wedge U_j$ has the
factor $U_j$ twice.\qed

\section{The formalisation}

The theorem is checked in Unit, in \code{sag/depth2}, with univalence
and replacement as parameters as in the rest of \code{sag}
(\code{./segc sag/depth2 all}: 170 modules, about 2.8\,s, every file
under 0.15\,s).  It uses the exterior algebra of \code{sag/exterior}
(products of generators, coordinates, the contraction $c_v$ by a vector
and the grade involution $\alpha$) and the convention of \code{sag/cffr}
that minors are exterior coordinates.  Grade is the explicit form above
(\code{Depth}); for a family indexed by $\{1,\dots,N\}$ it coincides with
the Koszul grade of \code{sag/koszul} (\code{DepthKoszul}).

Lemma~\ref{lem:key} is proved without signs $(-1)^j$, by induction on $n$
in the form $c_\varphi(\omega)\wedge b=-\alpha\bigl(\sum_j\varphi(U_j)P_j(b)\bigr)$,
using $c_\varphi(x\wedge w)=c_\varphi(x)\wedge w+\varphi(w)\,\alpha(x)$ and
the anticommutation of generators; then
$\sum_j\varphi(U_j)P_j(b)=\varphi(b)\,\omega-\alpha(c_\varphi(\omega\wedge
b))$.  The coefficients $k_{I'}$ in (2) come from expanding the linear form
$y\mapsto\co{-\alpha(c_\varphi(y))}I$ in the basis $(e_{I'})$.
